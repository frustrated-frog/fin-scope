**Tech Stack:** React 18、TypeScript、Vitest、Testing Library、CSS。

```ts
export function presentThesisStage(thesis: ResearchThesis, findingCount: number) {
  if (thesis.status === 'CONCLUDED') return { label: '已形成结论', tone: 'settled' as const };
  if (thesis.conclusion) return { label: '持续验证', tone: 'active' as const };
  if (findingCount > 0) return { label: '证据积累中', tone: 'active' as const };
  return { label: '待验证', tone: 'pending' as const };
}

export function presentConfidence(value?: ResearchThesis['confidence']) {
  return value === 'HIGH' ? '高置信' : value === 'MEDIUM' ? '中等置信' : value === 'LOW' ? '低置信' : '尚未评级';
}

export function groupThesisFindings(findings: ThesisFinding[], limit = 2) {
  return Object.fromEntries(['SUPPORT', 'COUNTER', 'UNKNOWN'].map((stance) => {
    const all = findings.filter((finding) => finding.stance === stance);
    return [stance, { items: all.slice(0, limit), total: all.length, remaining: Math.max(0, all.length - limit) }];
  }));
}
```

`summarizeResearchDiagnostics` 使用 `run.fetchedSourceCount`、`plannedSources.length` 和 `agentRuns.length` 生成固定结构。

      <header>阶段、置信度、当前判断和下一验证点</header>

      </div>
      <footer>{detail.runs.length} 次研究 · {detail.outputs.length} 项关联产物 · 最近研究日期</footer>
